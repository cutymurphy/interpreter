EPS = 0.000000001;

function sqrt(x) {
    if (x < 0.0) {
        return -1;
    }

    // float a = (x < 1) ? 0 : 1,
    //       b = (x < 1) ? 1 : x;
    a = 1.0;
    b = x;
    if (x < 1.0) {
        a = 0.0;
        b = 1.0;
    }

    while (b - a > EPS) {
        c = (a + b) / 2.0;
        if (c * c > x) {
            b = c;
        } else {
            a = c;
        }
    }

    return (a + b) / 2.0;
}

println(sqrt(2.0));
