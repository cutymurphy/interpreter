EPS = 0.000000001;

function sqrt(x) {
    if (x < 0.0) {
        return -1;
    }

    a = 1;
    b = x;
    if (x < 1.0) {
        a = 0;
        b = 1;
    }

    while (b - a > EPS) {
        c = (a + b) / 2.0;
        if (c * c > x) {
            b = c;
        } else {
            a = c;
        }
    }

    return (a + b) / 2.0;
}

println(sqrt(4.0));
println(sqrt(8.0));
println(sqrt(9.0));
